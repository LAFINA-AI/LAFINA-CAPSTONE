import subprocess,re,sys,collections
pats={
 'google_api_key':rb'AIza[0-9A-Za-z_\-]{35}',
 'openai_like_sk':rb'\bsk-[A-Za-z0-9_\-]{20,}',
 'github_token':rb'\bgh[pousr]_[A-Za-z0-9]{30,}',
 'private_key_block':rb'-----BEGIN (?:RSA |EC |OPENSSH |)PRIVATE KEY-----',
 'db_url_with_password':rb'(?:postgres(?:ql)?|mysql|mongodb(?:\+srv)?)://[^\s:/@"\']+:[^\s@"\']+@',
 'aws_access_key':rb'\bAKIA[0-9A-Z]{16}\b',
 'slack_token':rb'\bxox[baprs]-[A-Za-z0-9-]{10,}',
 'jwt':rb'\beyJ[A-Za-z0-9_-]{10,}\.eyJ[A-Za-z0-9_-]{10,}\.[A-Za-z0-9_-]{10,}',
 'pinecone_key':rb'\bpcsk_[A-Za-z0-9_]{20,}',
 'generic_assignment':rb'(?i)(?:api[_-]?key|secret|password|token)\s*[:=]\s*["\'][A-Za-z0-9_\-\.]{16,}["\']',
}
mode=sys.argv[1]
def blobs():
    if mode=='head':
        out=subprocess.run(['git','ls-tree','-r','HEAD'],capture_output=True).stdout.decode().splitlines()
        for l in out:
            meta,path=l.split('\t',1); _,t,sha=meta.split()
            if t=='blob': yield sha,path
    else:
        out=subprocess.run(['git','rev-list','--objects','--all'],capture_output=True).stdout.decode().splitlines()
        for l in out:
            parts=l.split(' ',1)
            if len(parts)==2: yield parts[0],parts[1]
hits=collections.defaultdict(set)
seen=set()
for sha,path in blobs():
    if sha in seen: continue
    seen.add(sha)
    t=subprocess.run(['git','cat-file','-t',sha],capture_output=True).stdout.strip()
    if t!=b'blob': continue
    size=int(subprocess.run(['git','cat-file','-s',sha],capture_output=True).stdout)
    if size>5_000_000: continue
    data=subprocess.run(['git','cat-file','blob',sha],capture_output=True).stdout
    for name,p in pats.items():
        if re.search(p,data): hits[name].add(path)
for k,v in hits.items():
    print(k, len(v)); [print('   ',x) for x in sorted(v)[:25]]
print('scanned blobs',len(seen))
